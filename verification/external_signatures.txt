def CubeSections.UpperBoundStatement :
  Prop :=
  forall (n k : Nat),
  LE.le (OfNat.ofNat 1) k → LE.le k n → (forall (U : CubeSections.Frame n k),
  LE.le (CubeSections.sectionVolume U) (CubeSections.sharpValue n k))

def CubeSections.EqualityStatement :
  Prop :=
  forall (n k : Nat),
  LE.le (OfNat.ofNat 1) k → LE.le k n → (forall (U : CubeSections.Frame n k),
  Iff (Eq (CubeSections.sectionVolume U) (CubeSections.sharpValue n k)) (CubeSections.IsBalancedSection U))

def CubeSections.AttainmentStatement :
  Prop :=
  forall (n k : Nat),
  LE.le (OfNat.ofNat 1) k → LE.le k n → Exists
    (fun (U : CubeSections.Frame n k) =>
    And (CubeSections.IsBalancedSection U) (Eq (CubeSections.sectionVolume U) (CubeSections.sharpValue n k)))

def CubeSections.MainStatement :
  Prop :=
  And CubeSections.UpperBoundStatement (And CubeSections.EqualityStatement CubeSections.AttainmentStatement)
